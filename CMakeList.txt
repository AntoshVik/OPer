cmake_minimum_required(VERSION 3.10)
project(CIFSMonitor)

set(CMAKE_CXX_STANDARD 17)

# Подключение nlohmann/json (можно через FetchContent или установить системно)
find_package(nlohmann_json REQUIRED)

add_executable(cifs-monitor
    main.cpp
    ConfigLoader.cpp
    LogMonitor.cpp
    ActionExecutor.cpp
)

target_link_libraries(cifs-monitor PRIVATE nlohmann_json::nlohmann_json)
